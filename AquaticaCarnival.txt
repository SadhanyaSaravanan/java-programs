import java.util.Scanner;
public class AquaticaCarnival
{
	public static void main(String[] args) {
	    Scanner sc=new Scanner(System.in);
		System.out.println("Enter the Hurl Factor,spin factor and speed factor:");
		int hurl=sc.nextInt();
	
		int spin=sc.nextInt();
		
		int speed=sc.nextInt();
		
		if(hurl>50&&spin>60&&speed>100){
		    System.out.println("GRADE 10");
		}
		else if(hurl>50&&spin>60){
		    System.out.println("GRADE 9");
		}
		else if(spin>60&&speed>100){
		    System.out.println("GRADE 8");
		}
		else if(hurl>50&&speed>100){
		    System.out.println("GRADE 7");
		}
		else if(hurl>50||spin>60||speed>100){
		    System.out.println("GRADE 6");
		}
		else{
		    System.out.println("GRADE 5");
		}
		
	}
	
}
